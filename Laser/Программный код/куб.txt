#include <SPI.h>
#include <nRF24L01.h>
#include <RF24.h>
#include <Servo.h>

#define CE_PIN 9
#define CSN_PIN 10
#define LASER_PIN 4
#define SERVO_H_PIN 6
#define SERVO_V_PIN 5
#define LED_PIN 13
#define BUZZER_PIN 3

// Команды
#define CMD_FULL_SHOW 1
#define CMD_EMERGENCY_STOP 99

RF24 radio(CE_PIN, CSN_PIN);
const byte address[6] = "00001";

Servo ServoH;
Servo ServoV;

// Глобальный флаг для экстренной остановки
volatile bool emergencyStopFlag = false;

void setup() {
  Serial.begin(9600);
  Serial.println("Cube Executor v3.0 - Emergency Stop Fixed");
  
  pinMode(LASER_PIN, OUTPUT);
  pinMode(LED_PIN, OUTPUT);
  pinMode(BUZZER_PIN, OUTPUT);
  
  digitalWrite(LASER_PIN, LOW);
  digitalWrite(LED_PIN, LOW);
  digitalWrite(BUZZER_PIN, LOW);

  ServoH.attach(SERVO_H_PIN);
  ServoV.attach(SERVO_V_PIN);
  
  ServoH.write(90);
  ServoV.write(90);
  delay(500);

  if (!radio.begin()) {
    Serial.println("Radio FAIL!");
    while (1);
  }

  radio.setChannel(76);
  radio.setDataRate(RF24_1MBPS);
  radio.setPALevel(RF24_PA_LOW);
  radio.openReadingPipe(1, address);
  radio.startListening();
  
  Serial.println("Cube Ready - Listening for commands");
  beep(2, 100);
}

void beep(int count, int duration) {
  for (int i = 0; i < count; i++) {
    tone(BUZZER_PIN, 1000, duration);
    delay(duration * 2);
  }
}

// Функция проверки экстренной остановки (вызывается часто!)
bool checkEmergencyStop() {
  if (emergencyStopFlag) {
    performEmergencyStop();
    return true;
  }
  return false;
}

// Выполнение экстренной остановки
void performEmergencyStop() {
  Serial.println("!!! PERFORMING EMERGENCY STOP !!!");
  
  // 1. Немедленно выключаем лазер
  digitalWrite(LASER_PIN, LOW);
  
  // 2. Возвращаем серво в безопасное положение
  ServoH.write(90);
  ServoV.write(90);
  delay(100);
  
  // 3. Звуковой и световой сигнал
  for (int i = 0; i < 3; i++) {
    digitalWrite(LED_PIN, HIGH);
    tone(BUZZER_PIN, 800, 300);
    delay(300);
    digitalWrite(LED_PIN, LOW);
    delay(300);
  }
  
  // 4. Длинный сигнал подтверждения
  tone(BUZZER_PIN, 600, 1000);
  delay(1000);
  noTone(BUZZER_PIN);
  
  // 5. Отправляем подтверждение остановки на пульт
  sendResponse(CMD_EMERGENCY_STOP);
  
  // 6. Сбрасываем флаг
  emergencyStopFlag = false;
  
  Serial.println("Emergency stop complete");
}

// Отправка ответа на пульт
void sendResponse(int response) {
  radio.stopListening();
  delay(5);
  radio.write(&response, sizeof(response));
  delay(5);
  radio.startListening();
  
  Serial.print("Response sent: ");
  Serial.println(response);
}

// Проверка входящих команд ВО ВРЕМЯ выполнения
void checkRadioDuringShow() {
  if (radio.available()) {
    int cmd = 0;
    radio.read(&cmd, sizeof(cmd));
    
    if (cmd == CMD_EMERGENCY_STOP) {
      Serial.println("Emergency stop command received DURING show!");
      emergencyStopFlag = true;
    }
  }
}

// СТАДИЯ с проверкой остановки
void stageHorizontal() {
  Serial.println("Stage 2: Horizontal scan");
  digitalWrite(LASER_PIN, HIGH);
  
  for (int angle = 50; angle <= 130; angle += 10) {
    ServoH.write(angle);
    
    // ПРОВЕРЯЕМ ОСТАНОВКУ НА КАЖДОМ ШАГЕ
    checkRadioDuringShow();
    if (checkEmergencyStop()) return;
    
    delay(250);
  }
  
  digitalWrite(LASER_PIN, LOW);
  delay(100);
  
  // Проверка перед отправкой ответа
  if (checkEmergencyStop()) return;
  
  ServoH.write(90);
  sendResponse(2);
}

void stageVertical() {
  Serial.println("Stage 3: Vertical scan");
  digitalWrite(LASER_PIN, HIGH);
  
  for (int angle = 50; angle <= 130; angle += 10) {
    ServoV.write(angle);
    
    // ПРОВЕРЯЕМ ОСТАНОВКУ НА КАЖДОМ ШАГЕ
    checkRadioDuringShow();
    if (checkEmergencyStop()) return;
    
    delay(250);
  }
  
  digitalWrite(LASER_PIN, LOW);
  delay(100);
  
  if (checkEmergencyStop()) return;
  
  ServoV.write(90);
  sendResponse(3);
}

void stageDiagonal1() {
  Serial.println("Stage 4: Diagonal scan 1");
  digitalWrite(LASER_PIN, HIGH);
  
  for (int i = 50; i <= 130; i += 10) {
    ServoH.write(i);
    ServoV.write(i);
    
    // ПРОВЕРЯЕМ ОСТАНОВКУ НА КАЖДОМ ШАГЕ
    checkRadioDuringShow();
    if (checkEmergencyStop()) return;
    
    delay(250);
  }
  
  digitalWrite(LASER_PIN, LOW);
  delay(100);
  
  if (checkEmergencyStop()) return;
  
  ServoH.write(90);
  ServoV.write(90);
  sendResponse(4);
}

void stageDiagonal2() {
  Serial.println("Stage 5: Diagonal scan 2");
  digitalWrite(LASER_PIN, HIGH);
  
  for (int i = 50; i <= 130; i += 10) {
    ServoH.write(i);
    ServoV.write(180 - i);
    
    // ПРОВЕРЯЕМ ОСТАНОВКУ НА КАЖДОМ ШАГЕ
    checkRadioDuringShow();
    if (checkEmergencyStop()) return;
    
    delay(250);
  }
  
  digitalWrite(LASER_PIN, LOW);
  delay(100);
  
  if (checkEmergencyStop()) return;
  
  ServoH.write(90);
  ServoV.write(90);
  
  // Финальный сигнал
  beep(3, 200);
  digitalWrite(LED_PIN, HIGH);
  delay(500);
  digitalWrite(LED_PIN, LOW);
  
  sendResponse(5);
}

// ПОЛНОЕ ШОУ с частой проверкой радио
void runFullShow() {
  Serial.println("=== STARTING FULL SHOW ===");
  
  // Убедимся, что флаг сброшен
  emergencyStopFlag = false;
  
  // Стадия 2
  stageHorizontal();
  if (emergencyStopFlag) return;
  
  // Проверяем радио между стадиями
  checkRadioDuringShow();
  if (checkEmergencyStop()) return;
  
  // Стадия 3
  stageVertical();
  if (emergencyStopFlag) return;
  
  checkRadioDuringShow();
  if (checkEmergencyStop()) return;
  
  // Стадия 4
  stageDiagonal1();
  if (emergencyStopFlag) return;
  
  checkRadioDuringShow();
  if (checkEmergencyStop()) return;
  
  // Стадия 5
  stageDiagonal2();
  
  Serial.println("=== SHOW COMPLETE ===");
}

void moveSingleServo(Servo &servo, const char* name) {
  Serial.print("Moving ");
  Serial.println(name);
  
  servo.write(50);
  delay(800);
  servo.write(130);
  delay(800);
  servo.write(90);
  delay(500);
}

void loop() {
  if (radio.available()) {
    int cmd = 0;
    radio.read(&cmd, sizeof(cmd));
    Serial.print("Command received: ");
    Serial.println(cmd);
    
    // Немедленная обработка команды экстренной остановки
    if (cmd == CMD_EMERGENCY_STOP) {
      Serial.println("Emergency stop command received");
      emergencyStopFlag = true;
      return;
    }
    
    // Если уже в аварийном режиме - игнорируем другие команды
    if (emergencyStopFlag) {
      Serial.println("Ignoring command - emergency stop active");
      return;
    }
    
    // Обработка обычных команд
    if (cmd == CMD_FULL_SHOW) {
      runFullShow();
    } 
    else if (cmd == 2) {
      moveSingleServo(ServoH, "Servo Horizontal");
      sendResponse(2);
    } 
    else if (cmd == 3) {
      moveSingleServo(ServoV, "Servo Vertical");
      sendResponse(3);
    } 
    else if (cmd == 4) {
      digitalWrite(LASER_PIN, HIGH);
      delay(1000);
      digitalWrite(LASER_PIN, LOW);
      sendResponse(4);
    }
    else if (cmd == 5) {
      // Долгое мигание с проверкой остановки
      for (int i = 0; i < 10; i++) {
        digitalWrite(LED_PIN, HIGH);
        delay(500);
        digitalWrite(LED_PIN, LOW);
        delay(500);
        
        checkRadioDuringShow();
        if (emergencyStopFlag) {
          performEmergencyStop();
          return;
        }
      }
      sendResponse(5);
    }
  }
  
  // Индикация работы
  static unsigned long lastBlink = 0;
  if (millis() - lastBlink > 2000) {
    digitalWrite(LED_PIN, !digitalRead(LED_PIN));
    lastBlink = millis();
  }
}
