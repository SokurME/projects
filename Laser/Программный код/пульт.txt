#include <SPI.h>
#include <nRF24L01.h>
#include <RF24.h>
#include <Wire.h> 
#include <LiquidCrystal_I2C.h>

#define CE_PIN 9
#define CSN_PIN 10 
#define BTN_SEND 2     
#define ENCODER_S1 6   
#define ENCODER_S2 4   
#define ENCODER_KEY 5  

// Команды
#define CMD_FULL_SHOW 1
#define CMD_EMERGENCY_STOP 99

RF24 radio(CE_PIN, CSN_PIN);
const byte address[6] = "00001";
LiquidCrystal_I2C lcd(0x27, 16, 2);

// Состояния
enum State {
  STATE_MENU,
  STATE_SHOW_RUNNING,
  STATE_EMERGENCY_STOP
};

State currentState = STATE_MENU;

// Переменные
int menuCounter = 1;   
bool lastClk;
bool btnSendPressed = false;
bool encoderKeyPressed = false;
unsigned long sendTime = 0;
unsigned long keyHoldStart = 0;
bool keyHolding = false;
const unsigned long HOLD_TIME_FOR_STOP = 1000; // Удержание 1 сек для остановки
const unsigned long RESPONSE_TIMEOUT = 15000; // 15 сек таймаут

// Меню
const char* menuItems[] = {
  "LASER           ",
  "SERVO 1 (H)     ",
  "SERVO 2 (V)     ",
  "MICROCONTROLLER "
};

void setup() {
  pinMode(BTN_SEND, INPUT_PULLUP);
  pinMode(ENCODER_S1, INPUT_PULLUP);
  pinMode(ENCODER_S2, INPUT_PULLUP);
  pinMode(ENCODER_KEY, INPUT_PULLUP);
  
  Serial.begin(9600);
  Serial.println("Pult Remote v2.0 with Emergency Stop");
  
  lcd.init();
  lcd.backlight();
  lcd.clear();
  
  if (!radio.begin()) {
    lcd.setCursor(0, 0);
    lcd.print("Radio FAIL!");
    while (1);
  }
  
  radio.openWritingPipe(address);
  radio.openReadingPipe(1, address);
  radio.setPALevel(RF24_PA_MAX);
  radio.setDataRate(RF24_1MBPS);
  radio.setChannel(76);
  radio.stopListening();
  
  lastClk = digitalRead(ENCODER_S1);
  updateDisplay();
}

// Обновление дисплея в режиме меню
void updateDisplay() {
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("ADR:00001 MODE:");
  lcd.setCursor(0, 1);
  lcd.print(">");
  lcd.print(menuItems[menuCounter - 1]);
}

// Обновление дисплея во время шоу
void updateShowDisplay(const char* line1, const char* line2) {
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print(line1);
  lcd.setCursor(0, 1);
  lcd.print(line2);
}

// Отправка команды
bool sendCommand(int cmd) {
  radio.stopListening();
  bool success = radio.write(&cmd, sizeof(cmd));
  radio.startListening();
  return success;
}

// Обработка энкодера
void handleEncoder() {
  bool currentClk = digitalRead(ENCODER_S1);
  
  if (currentClk != lastClk && currentClk == LOW) {
    if (currentState == STATE_MENU) {
      if (digitalRead(ENCODER_S2) != currentClk) {
        menuCounter++;
        if (menuCounter > 4) menuCounter = 1;
      } else {
        menuCounter--;
        if (menuCounter < 1) menuCounter = 4;
      }
      updateDisplay();
    }
  }
  lastClk = currentClk;
}

// Обработка кнопок
void handleButtons() {
  // Кнопка отправки (красная)
  bool sendBtnState = digitalRead(BTN_SEND) == LOW;
  if (sendBtnState && !btnSendPressed) {
    btnSendPressed = true;
    
    if (currentState == STATE_MENU) {
      // Отправляем команду в зависимости от выбранного пункта меню
      int cmdToSend = menuCounter;
      
      if (cmdToSend == 1) { // Запуск полного шоу
        if (sendCommand(CMD_FULL_SHOW)) {
          currentState = STATE_SHOW_RUNNING;
          sendTime = millis();
          updateShowDisplay("RUN 00001: SCAN", "STOP:Hold Key 1s");
          Serial.println("Full show started");
        } else {
          updateShowDisplay("SEND ERROR!", "Retry...");
          delay(1000);
          updateDisplay();
        }
      } else { // Отправка одиночной команды
        if (sendCommand(cmdToSend)) {
          updateShowDisplay("Command sent", menuItems[menuCounter - 1]);
          delay(1000);
          updateDisplay();
        } else {
          updateShowDisplay("SEND ERROR!", "Retry...");
          delay(1000);
          updateDisplay();
        }
      }
    }
  }
  
  if (!sendBtnState) {
    btnSendPressed = false;
  }
  
  // Кнопка энкодера (для экстренной остановки)
  bool encoderBtnState = digitalRead(ENCODER_KEY) == LOW;
  
  if (encoderBtnState && !encoderKeyPressed) {
    encoderKeyPressed = true;
    keyHoldStart = millis();
    keyHolding = true;
  }
  
  if (!encoderBtnState && encoderKeyPressed) {
    encoderKeyPressed = false;
    keyHolding = false;
    
    // Короткое нажатие в режиме меню
    if (currentState == STATE_MENU && millis() - keyHoldStart < 300) {
      // Можно добавить действие на короткое нажатие
      updateShowDisplay("Short press", "in menu mode");
      delay(500);
      updateDisplay();
    }
  }
  
  // Проверка удержания кнопки энкодера
  if (keyHolding && encoderKeyPressed) {
    unsigned long holdDuration = millis() - keyHoldStart;
    
    if (holdDuration >= HOLD_TIME_FOR_STOP) {
      // АКТИВАЦИЯ ЭКСТРЕННОЙ ОСТАНОВКИ
      if (currentState == STATE_SHOW_RUNNING) {
        if (sendCommand(CMD_EMERGENCY_STOP)) {
          currentState = STATE_EMERGENCY_STOP;
          updateShowDisplay("EMERGENCY STOP!", "Signal sent");
          Serial.println("Emergency stop activated");
          
          // Сброс состояния через 2 секунды
          delay(2000);
          currentState = STATE_MENU;
          updateDisplay();
        } else {
          updateShowDisplay("STOP FAILED!", "Radio error");
          delay(2000);
          updateShowDisplay("RUN 00001: SCAN", "STOP:Hold Key 1s");
        }
      }
      keyHolding = false;
    } else if (currentState == STATE_SHOW_RUNNING) {
      // Показываем обратный отсчет до остановки
      int countdown = (HOLD_TIME_FOR_STOP - holdDuration) / 100;
      if (countdown < 10) {
        char line2[17];
        snprintf(line2, 17, "STOP:Hold %d..  ", countdown);
        updateShowDisplay("RUN 00001: SCAN", line2);
      }
    }
  }
}

// Обработка ответов от куба
void handleResponses() {
  if (currentState == STATE_SHOW_RUNNING) {
    if (radio.available()) {
      int response = 0;
      radio.read(&response, sizeof(response));
      sendTime = millis(); // Сброс таймера при получении ответа
      
      Serial.print("Response received: ");
      Serial.println(response);
      
      switch(response) {
        case 2:
          updateShowDisplay("HORIZONTAL SCAN", "Angle: 40,0 deg");
          break;
        case 3:
          updateShowDisplay("VERTICAL SCAN", "Angle: 0,40 deg");
          break;
        case 4:
          updateShowDisplay("DIAGONAL SCAN 1", "Angle: 40,40 deg");
          break;
        case 5:
          updateShowDisplay("SHOW COMPLETE!", "All stages done");
          delay(2000);
          currentState = STATE_MENU;
          updateDisplay();
          break;
      }
    }
    
    // Проверка таймаута
    if (millis() - sendTime > RESPONSE_TIMEOUT) {
      updateShowDisplay("TIMEOUT ERROR!", "No response");
      Serial.println("Response timeout");
      delay(2000);
      currentState = STATE_MENU;
      updateDisplay();
    }
  }
}

void loop() {
  // Всегда обрабатываем энкодер (меню меняется только в режиме меню)
  handleEncoder();
  
  // Всегда обрабатываем кнопки
  handleButtons();
  
  // Обработка ответов от куба (только во время шоу)
  handleResponses();
  
  // Мигание LED на плате во время шоу
  static unsigned long lastBlink = 0;
  if (currentState == STATE_SHOW_RUNNING && millis() - lastBlink > 500) {
    digitalWrite(LED_BUILTIN, !digitalRead(LED_BUILTIN));
    lastBlink = millis();
  } else if (currentState != STATE_SHOW_RUNNING) {
    digitalWrite(LED_BUILTIN, LOW);
  }
}
